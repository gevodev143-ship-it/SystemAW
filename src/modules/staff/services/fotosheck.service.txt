import { supabase } from "../../../lib/supabase";

const obtenerNombreBaseImagen = (nombreArchivo: string): string => {
  const idx = nombreArchivo.lastIndexOf(".");
  return idx === -1 ? nombreArchivo : nombreArchivo.slice(0, idx);
};

const construirRutaFotosheck = (stffLinkImg: string): string => {
  const nombreBase = obtenerNombreBaseImagen(stffLinkImg);
  return `gestion_empresarial/personal_fotosheck/fotosheck${nombreBase}.webp`;
};

const construirNombreArchivo = (stffLinkImg: string): string => {
  const nombreBase = obtenerNombreBaseImagen(stffLinkImg);
  return `fotosheck${nombreBase}.webp`;
};

export const existeFotosheck = async (
  custNameBucket: string,
  stffLinkImg: string
): Promise<boolean> => {
  const carpeta = "gestion_empresarial/personal_fotosheck";
  const nombreArchivo = construirNombreArchivo(stffLinkImg);

  const { data, error } = await supabase.storage
    .from(custNameBucket)
    .list(carpeta);

  if (error) {
    console.error("Error al listar fotosheck:", error);
    return false;
  }

  return (data ?? []).some((archivo) => archivo.name === nombreArchivo);
};

export const obtenerUrlFotosheck = (custNameBucket: string, stffLinkImg: string): string => {
  const ruta = construirRutaFotosheck(stffLinkImg);
  const { data } = supabase.storage.from(custNameBucket).getPublicUrl(ruta);
  return data.publicUrl;
};

export const subirFotosheck = async (
  custNameBucket: string,
  stffLinkImg: string,
  blobWebp: Blob
): Promise<string> => {
  const ruta = construirRutaFotosheck(stffLinkImg);

  const { error } = await supabase.storage
    .from(custNameBucket)
    .upload(ruta, blobWebp, { contentType: "image/webp", upsert: true });

  if (error) throw error;

  return obtenerUrlFotosheck(custNameBucket, stffLinkImg);
};
