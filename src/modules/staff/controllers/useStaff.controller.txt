import { useEffect, useState } from "react";
import { useAuth } from "../../../core/contexts/auth.context"; //para traer el JWT el id del cust_id
import type { Staff } from "../types/staff.type";
import {
  obtenerStaffsPorCustId,
  actualizarEstadoStaff,
  eliminarStaff,
} from "../services/staff.service";

const ITEMS_POR_PAGINA = 5;

export const useStaffController = () => {
  const { custId } = useAuth();

  const [staffs, setStaffs] = useState<Staff[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");
  const [menuAbierto, setMenuAbierto] = useState<number | null>(null);
  const [paginaActual, setPaginaActual] = useState(1);

  const fetchStaffs = async (id: number) => {
    setLoading(true);
    setError("");

    const { data, error } = await obtenerStaffsPorCustId(id);

    if (error) {
      console.error("ERROR AL OBTENER PERSONALES:", error);
      setError("No se pudieron cargar los personales.");
    } else {
      setStaffs(data ?? []);
    }

    setLoading(false);
  };

  useEffect(() => {
    if (custId !== null) fetchStaffs(custId);
  }, [custId]);

  const toggleActivo = async (staff: Staff) => {
    if (!custId) return;

    const { error } = await actualizarEstadoStaff(staff, custId);

    if (error) {
      console.error("ERROR AL CAMBIAR ESTADO:", error);
      alert("No se pudo cambiar el estado del personal.");
      return;
    }

    setMenuAbierto(null);
    await fetchStaffs(custId);
  };

  const handleEliminar = async (staff: Staff) => {
    if (!custId) return;

    const confirmar = window.confirm(
      `¿Seguro que deseas eliminar a "${staff.stff_name} ${staff.stff_lastname}"?`
    );
    if (!confirmar) return;

    const { error } = await eliminarStaff(staff, custId);

    if (error) {
      console.error("ERROR AL ELIMINAR PERSONAL:", error);
      alert("No se pudo eliminar el personal.");
      return;
    }

    setMenuAbierto(null);
    await fetchStaffs(custId);
  };

  const totalPaginas = Math.max(1, Math.ceil(staffs.length / ITEMS_POR_PAGINA));
  const staffsPagina = staffs.slice(
    (paginaActual - 1) * ITEMS_POR_PAGINA,
    paginaActual * ITEMS_POR_PAGINA
  );

  return {
    staffs,
    staffsPagina,
    loading,
    error,
    menuAbierto,
    setMenuAbierto,
    paginaActual,
    setPaginaActual,
    totalPaginas,
    itemsPorPagina: ITEMS_POR_PAGINA,
    toggleActivo,
    handleEliminar,
  };
};
